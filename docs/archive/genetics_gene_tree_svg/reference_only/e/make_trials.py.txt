from pathlib import Path
from html import escape
import subprocess
import xml.etree.ElementTree as ET

BASE=Path('_temp/genetics_visual_renewal/m3_phylogenetics/e')
BLUE='#006CA8'; RED='#B54400'; INK='#222222'; GREEN='#007A52'; GOLD='#F2C94C'

def text(x,y,t,size=24,color=INK,anchor='start'):
    return f'<text x="{x}" y="{y}" font-size="{size}" fill="{color}" text-anchor="{anchor}">{escape(t)}</text>'
def path(d,color=INK,width=3,dash=''):
    return f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{width}" stroke-linejoin="round" stroke-linecap="round"'+(f' stroke-dasharray="{dash}"' if dash else '')+'/>'
def line(x1,y1,x2,y2,color=INK,width=3,dash=''):
    return path(f'M{x1},{y1} L{x2},{y2}',color,width,dash)
def cross(x,y,color=INK):
    return path(f'M{x-7},{y-7} L{x+7},{y+7} M{x-7},{y+7} L{x+7},{y-7}',color,3)
def save(name,w,h,title,desc,body):
    svg=f'<svg xmlns="http://www.w3.org/2000/svg" width="{w}" height="{h}" viewBox="0 0 {w} {h}" role="img" aria-labelledby="title description"><title id="title">{escape(title)}</title><desc id="description">{escape(desc)}</desc><g font-family="Arial, sans-serif">'+body+'</g></svg>'
    p=BASE/'assets'/name;p.write_text(svg)
    root=ET.parse(p).getroot(); assert not root.findall('.//{http://www.w3.org/2000/svg}image')
    for width in [320,1200]:
        subprocess.run(['rsvg-convert','-b','white','-w',str(width),'-o',str(BASE/'previews'/f'{p.stem}_{width}.png'),str(p)],check=True)

def locus(x,y,lost=(),focus=None,scale=1):
    s=f'<g transform="translate({x} {y}) scale({scale})">'
    s+='<rect x="0" y="0" width="310" height="22" fill="#ABC2CC"/>'
    for i,tissue in enumerate(slots):
        color='#404040' if i in lost else (GOLD if tissue==1 else GREEN)
        s+=f'<rect x="{25+i*34}" y="0" width="12" height="22" fill="{color}"/>'
        if i in lost:s+=path(f'M{25+i*34},-6 L{37+i*34},28',INK,2)
    s+='<rect x="245" y="0" width="25" height="22" fill="white" stroke="#222" stroke-width="2"/><rect x="270" y="0" width="80" height="22" fill="#C84458"/>'
    s+=path('M270,31 V44 H330 M321,37 L331,44 L321,51',INK,2)
    if focus is not None:s+=path(f'M{31+focus*34},-33 V-9 M{25+focus*34},-16 L{31+focus*34},-8 L{37+focus*34},-16',RED,3)
    return s+'</g>'

def edges(path):
    tokens=re.findall(r'[MLHV]|-?\d+(?:\.\d+)?',path)
    out=[];i=0;point=None
    while i<len(tokens):
        cmd=tokens[i];i+=1
        if cmd in 'ML': new=(float(tokens[i]),float(tokens[i+1]));i+=2
        elif cmd=='H': new=(float(tokens[i]),point[1]);i+=1
        elif cmd=='V': new=(point[0],float(tokens[i]));i+=1
        else: raise AssertionError(cmd)
        if cmd!='M':out.append((point,new))
        point=new
    return out

def topology(name,root,leaves,expected):
    r=ET.parse(ASSETS/name).getroot()
    g=r.find('.//s:g[@id="gene-tree"]',NS)
    graph={}
    for path in g.findall('s:path',NS):
        for a,b in edges(path.attrib['d']):graph.setdefault(a,[]).append(b)
    clades=[]
    def descendants(point):
        if point in leaves:return {leaves[point]}
        assert point in graph,point
        result=set().union(*(descendants(q) for q in graph[point]))
        if len(result)>1:clades.append(sorted(result))
        return result
    assert descendants(root)==set(leaves.values())
    for expected_clade in expected:assert sorted(expected_clade) in clades
    checks[name]['reconstructed_clades']=clades

from pathlib import Path
from html import escape
import xml.etree.ElementTree as ET
import cairosvg
P=Path(__file__).parent

def tree(name,tips,merges,stage,scale,source):
    ys={t:55+i*75 for i,t in enumerate(tips)}
    hs={t:0 for t in tips}
    out=['<svg xmlns="http://www.w3.org/2000/svg" width="860" height="570" viewBox="0 0 860 570" role="img" aria-labelledby="title desc">',f'<title id="title">{escape(name)}</title>',f'<desc id="desc">Metric tree. Horizontal branch lengths are proportional to distance. Source: {escape(source)}. Stage {stage}.</desc>','<rect width="860" height="570" fill="white"/>','<g font-family="Arial" font-size="27" fill="#172B3A">']
    x=lambda h:735-h*scale
    for i,(node,a,b,h) in enumerate(merges):
        ys[node]=(ys[a]+ys[b])/2; hs[node]=h
        if i>=stage:continue
        out.append(f'<g id="join-{node}"><path d="M {x(h)} {ys[a]} V {ys[b]}" stroke="#006A69" stroke-width="4" fill="none"/>')
        for t in (a,b):
            out.append(f'<path id="edge-{node}-{t}" d="M {x(h)} {ys[t]} H {x(hs[t])}" stroke="#006A69" stroke-width="4" fill="none"/>')
            out.append(f'<text x="{(x(h)+x(hs[t]))/2}" y="{ys[t]-12}" text-anchor="middle">{h-hs[t]:g}</text>')
        out.append(f'<circle cx="{x(h)}" cy="{ys[node]}" r="5" fill="#006A69"/><text x="{x(h)-12}" y="{ys[node]+9}" text-anchor="end">{escape(node)}</text></g>')
    visible=set()
    for _,a,b,_ in merges[:stage]:visible.update((a,b))
    for t in tips:
        if t in visible:out.append(f'<text x="750" y="{ys[t]+9}">{escape(t)}</text>')
    out+=['</g></svg>']
    path=P/(name+'.drawn.svg');path.write_text('\n'.join(out));ET.parse(path)
    for w in (320,1200):cairosvg.svg2png(url=str(path),write_to=str(P/f'{name}_{w}.png'),output_width=w)
    return path
